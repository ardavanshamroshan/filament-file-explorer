Implement FileExplorerAuthorizer (or generate one) to gate browse / upload / rename / delete per owner model.

Authorizer.php php 
final class QuestionFileExplorerAuthorizer implements FileExplorerAuthorizer
{{
    public function canBrowse(Model $owner, ?Authenticatable $user = null): bool
    {{
        return $user?->can('view', $owner) ?? false;
    }}
    // canUpload, canRename, canDelete, ...
}}
